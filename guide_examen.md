    try:
        resultat = (param1 * param2) / (param1 - param2)
    except ZeroDivisionError:
        print('Erreur : Division par zéro évitée.')
        return 0
    except TypeError:
        print('Erreur : Les paramètres doivent être numériques.')
        return None

    # Validation finale
    if resultat > 10:
        return resultat
    return -1
